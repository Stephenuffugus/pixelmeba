  phase('Repair')
  const keys = ['code-minors-6b', 'docs-a11y-6b'].filter((k) => open.some((p) => p.includes(`[${k}]`)))
  log(`repair: ${keys.join(', ') || 'lead-only problems'}`)
  const fixes = await parallel(keys.map((k) => () => ag(`${k === 'code-minors-6b' ? FIX_A : FIX_B}\n\nYou are the REPAIR round for this assignment (the work is in the tree; do not start over).
