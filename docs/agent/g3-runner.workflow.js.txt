const FINISH_NOTE = (what) => `IMPORTANT — RESUMING INTERRUPTED WORK. A previous builder with exactly this assignment was cut off; its uncommitted work is in the working tree (git status / git diff). ${what} Do NOT start over and do not discard its work. First audit the existing work against EVERY item of the assignment's Build and "Done when" lists: per item DONE (file:line + the test that proves it, re-run now) / PARTIAL / MISSING / WRONG. Then complete everything PARTIAL/MISSING/WRONG and re-run all your tests.`

const REPORT = (key, suffix) => `When you finish, write your final report ALSO to ${R}/${key}-${suffix}.md (create the folder if needed; this is the only docs file you may create), then return the same report as your final message.`

const BUILD_FINAL = `Final message: a plain report — a checklist of every Build and "Done when" item (DONE with file:line and the proving test / NOT DONE and why), files created/changed, what each test proves (and how you checked it fails without the mechanism), commands with results (paste the pass/fail summary lines), FENCE notes, bugs noticed elsewhere, proposed decisions.`

const lensPort = (t, lens) => t.port + (lens === 'rules' ? 5 : 75)

const VERIFY = (t, built, lens) => `You are an adversarial verifier for Pixelmeba (/workspaces/pixelmeba; follow CLAUDE.md). Do NOT edit any repository file except your own report file named below, and do NOT run git add/commit/stash/checkout/reset/restore. Other agents may still be editing other files. Scratch goes in tmp/verify-${t.key}-${lens}/ (git-ignored). 2 CPUs shared: targeted commands, generous timeouts, run_in_background + until-loop for long runs. Playwright: E2E_PORT=${lensPort(t, lens)} E2E_OUTDIR=tmp/dist-verify-${t.key}-${lens}; stop only that port before and after.

A builder was given this assignment:
---
${t.prompt}
---
The builder reported:
---
${typeof built === 'string' ? built : JSON.stringify(built)}
---
${LENSES[lens]}
Try to REFUTE that the assignment is done correctly, completely and honestly — every item of its Build and "Done when" lists. Return problems with file:line and a concrete repro, most severe first, each prefixed BLOCKER / MAJOR / MINOR; add "VERIFIED OK — <item> — <evidence>" entries for parts that hold. ok=false if any BLOCKER or MAJOR. Write the same list to ${R}/${t.key}-verify-${lens}.md.`

const FIX = (t, problems, round) => `${PRE}

Your Playwright port is ${t.port}.

You are the round-${round} fixer for this assignment (the work is in the tree; do not start over):
---
${t.prompt}
---
Adversarial verifiers found these problems (reports in ${R}/${t.key}-verify-*.md${round > 1 ? ` and ${R}/${t.key}-reverify${round - 1}-*.md` : ''}):
---
${problems.join('\n')}
---
Fix EVERY BLOCKER and MAJOR, and every MINOR that is cheap and within your files. For each behavioural problem add or tighten a test that fails on the old behaviour and passes now. If a problem is wrong (the verifier misread a rule), say why with quotes from the canonical docs instead of changing code. If a fix needs a file outside your ownership, make it only if it is small and additive, or describe it precisely for the lead. Re-run your unit tests, the fence fixtures and the task's e2e specs (all three projects) before finishing. ${REPORT(t.key, `fix${round}`)} Final message: per problem FIXED (how, which test) / NOT A PROBLEM (why, quotes) / NOT FIXED (why); files changed; commands + results.`

const REVERIFY = (t, fixReport, problems, round, lens) => `You are an adversarial re-verifier for Pixelmeba (/workspaces/pixelmeba; follow CLAUDE.md). Do NOT edit any repository file except your report file, and do NOT run git add/commit/stash/checkout/reset/restore. Other agents may still be editing other files. Scratch in tmp/reverify${round}-${t.key}-${lens}/. 2 CPUs: targeted commands. Playwright: E2E_PORT=${lensPort(t, lens)} E2E_OUTDIR=tmp/dist-reverify-${t.key}-${lens}; stop only that port before and after.

Original assignment:
---
${t.prompt}
---
Verifiers found these problems:
---
${problems.join('\n')}
---
A fixer (round ${round}) reported:
---
${fixReport}
---
${LENSES[lens]}
For EACH problem try to REFUTE that it is fixed: reproduce the original finding against the current tree, read the diff, check the new test fails on the old behaviour, and check nothing regressed (re-run the touched unit tests, the fence fixtures, and the task's e2e specs). Return remaining/new problems with file:line and repro, prefixed BLOCKER / MAJOR / MINOR, and "VERIFIED OK — <problem> — <evidence>" for fixed ones. ok=false if any BLOCKER or MAJOR remains. Write the same list to ${R}/${t.key}-reverify${round}-${lens}.md.`

const VERDICT = {
  type: 'object',
  properties: { ok: { type: 'boolean' }, problems: { type: 'array', items: { type: 'string' } } },
  required: ['ok', 'problems'],
}
const serious = (v) => (v && Array.isArray(v.problems) ? v.problems.filter((p) => /^(BLOCKER|MAJOR|MINOR)/.test(p.trim())) : [])
const blocking = (ps) => ps.filter((p) => /^(BLOCKER|MAJOR)/.test(p.trim()))

log(`${WAVE}: ${TASKS.length} task(s): ${TASKS.map((t) => t.key).join(', ')}`)

return await pipeline(
  TASKS,
  (t) =>
    agent(`${PRE}\n\nYour Playwright port is ${t.port}.\n\n${t.finish ? FINISH_NOTE(t.finish) + '\n\n' : ''}${t.prompt}\n\n${REPORT(t.key, 'build')} ${BUILD_FINAL}`, {
      label: `${t.finish ? 'finish' : 'build'}:${t.key}`,
      phase: 'Build',
    }),
  async (built, t) => {
    const lenses = t.lenses && t.lenses.length ? t.lenses : ['rules']
    const verdicts = await parallel(lenses.map((l) => () => agent(VERIFY(t, built, l), { label: `verify:${t.key}:${l}`, phase: 'Verify', schema: VERDICT })))
    return { built, lenses, verdicts: verdicts.filter(Boolean) }
  },
  async (v, t) => {
    const rounds = []
    let problems = v.verdicts.flatMap(serious)
    let ok = v.verdicts.length > 0 && v.verdicts.every((x) => x.ok) && blocking(problems).length === 0
    for (let round = 1; round <= 2 && problems.length > 0; round++) {
      const fixReport = await agent(FIX(t, problems, round), { label: `fix${round}:${t.key}`, phase: 'Fix' })
      const rv = (await parallel(v.lenses.map((l) => () => agent(REVERIFY(t, String(fixReport), problems, round, l), { label: `reverify${round}:${t.key}:${l}`, phase: 'Fix', schema: VERDICT })))).filter(Boolean)
      rounds.push({ round, fixReport, reverify: rv })
      if (!rv.length) break
      const remaining = rv.flatMap(serious)
      ok = rv.every((x) => x.ok) && blocking(remaining).length === 0
      problems = remaining
      if (ok) break
    }
    log(`${t.key}: ${ok ? 'clean' : 'problems remain'} after ${rounds.length} fix round(s); ${problems.length} open item(s), ${blocking(problems).length} blocking`)
    return { key: t.key, built: v.built, verdicts: v.verdicts, rounds, ok, remaining: problems }
  },
)
