async function runTask(t) {
  const built = await agent(`${PRE}\n\n${t.port ? `Your Playwright port is ${t.port}.` : 'Do not run Playwright.'}\n\n${t.prompt}\n\nWhen you finish, write your final report to ${R}/${t.key}-build.md (create the folder if needed) and return the same report: files changed, what each test proves, commands + results (paste the pass/fail summary lines), anything not done + why, bugs noticed elsewhere, proposed decisions.`, { label: `build:${t.key}`, phase: 'Build' })
  const verdicts = (await parallel(LENSES[t.key].map((l) => () => agent(VERIFY(t, built, l), { label: `verify:${t.key}:${l.key}`, phase: 'Verify', schema: VERDICT })))).filter(Boolean)
  let problems = verdicts.flatMap(serious)
  let ok = verdicts.length > 0 && verdicts.every((v) => v.ok) && blocking(problems).length === 0
  const rounds = []
  for (let round = 1; round <= 2 && problems.length > 0; round++) {
    const fixReport = await agent(FIX(t, problems, round), { label: `fix${round}:${t.key}`, phase: 'Fix' })
    const rv = await agent(REVERIFY(t, String(fixReport), problems, round), { label: `reverify${round}:${t.key}`, phase: 'Fix', schema: VERDICT })
    rounds.push({ round, fixReport, reverify: rv })
    if (!rv) break
    problems = serious(rv)
    ok = rv.ok && blocking(problems).length === 0
    if (ok) break
  }
  log(`${t.key}: ${ok ? 'clean' : 'problems remain'} after ${rounds.length} fix round(s)`)
  return { key: t.key, built, verdicts, rounds, ok, remaining: problems }
}
