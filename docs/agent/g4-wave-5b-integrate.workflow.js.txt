const VERDICT = {
  type: 'object',
  properties: { ok: { type: 'boolean' }, problems: { type: 'array', items: { type: 'string' } } },
  required: ['ok', 'problems'],
}

phase('Fix')
const [repA, repB] = await parallel([
  () => ag(FIX_A, { label: 'fix:objectives-minors', phase: 'Fix', effort: 'high' }),
  () => ag(FIX_B, { label: 'fix:stories-minors', phase: 'Fix', effort: 'high' }),
])
const reports = `=== FIXER objectives-minors ===\n${repA ? String(repA) : 'NO REPORT'}\n\n=== FIXER stories-minors ===\n${repB ? String(repB) : 'NO REPORT'}`

Write the same list, plus every command you ran with its summary line, to the report file. ok=true only if nothing BLOCKER/MAJOR remains.`

phase('Verify')
let verdict = await ag(VERIFY(1), { label: 'verify:integrate-5b', phase: 'Verify', schema: VERDICT, effort: 'high' })
const blockingOf = (v) => (v && Array.isArray(v.problems) ? v.problems.filter((p) => /^\s*(BLOCKER|MAJOR)/.test(p)) : [])
let open = blockingOf(verdict)
const rounds = []
if (open.length) {
  phase('Repair')
  const keys = ['objectives-minors', 'stories-minors'].filter((k) => open.some((p) => p.includes(`[${k}]`)))
  log(`repair: ${keys.join(', ') || 'lead-only problems'}`)
  const fixes = await parallel(keys.map((k) => () => ag(`${k === 'objectives-minors' ? FIX_A : FIX_B}\n\nYou are the REPAIR round for this assignment (the work is in the tree; do not start over). The adversarial verifier (docs/reports/reviews/g4-wave-5b/integrate-verify1.md) found:\n${open.filter((p) => p.includes(`[${k}]`)).join('\n')}\nFix each, prove each with a test that fails without the fix, and report as before (write the report to docs/reports/reviews/g4-wave-5b/integrate-repair-${k}.md).`, { label: `repair:${k}`, phase: 'Repair', effort: 'high' })))
  rounds.push({ keys, fixes: fixes.map((f) => (f ? String(f) : null)) })
  if (keys.length) {
    verdict = await ag(`${VERIFY(2)}\n\nREPAIR ROUND: the earlier verifier's BLOCKER/MAJOR problems were:\n${open.join('\n')}\nRepair reports:\n${fixes.map((f, i) => `=== ${keys[i]} ===\n${f ? String(f) : 'NO REPORT'}`).join('\n\n')}\nFor EACH earlier problem, reproduce it against the current tree and say VERIFIED FIXED or NOT FIXED; then grade anything new as before.`, { label: 'reverify:integrate-5b', phase: 'Repair', schema: VERDICT, effort: 'high' })
    open = blockingOf(verdict)
  }
}
log(`integration 5b: ${open.length ? `${open.length} BLOCKER/MAJOR remain` : 'no BLOCKER/MAJOR remains'}`)
return { reports: { objectivesMinors: repA ? String(repA) : null, storiesMinors: repB ? String(repB) : null }, verdict, rounds, open }
