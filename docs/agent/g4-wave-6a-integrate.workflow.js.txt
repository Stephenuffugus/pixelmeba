phase('Verify')
let verdict = await ag(VERIFY(1), { label: 'verify:integrate-6a', phase: 'Verify', schema: VERDICT, effort: 'high' })
const blockingOf = (v) => (v && Array.isArray(v.problems) ? v.problems.filter((p) => /^\s*(BLOCKER|MAJOR)/.test(p)) : [])
let open = blockingOf(verdict)
const rounds = []
if (open.length) {
  phase('Repair')
  const keys = ['code-minors-6a', 'docs-a11y-6a'].filter((k) => open.some((p) => p.includes(`[${k}]`)))
  log(`repair: ${keys.join(', ') || 'lead-only problems'}`)
  const fixes = await parallel(keys.map((k) => () => ag(`${k === 'code-minors-6a' ? FIX_A : FIX_B}\n\nYou are the REPAIR round for this assignment (the work is in the tree; do not start over). The adversarial verifier (docs/reports/reviews/g4-wave-6a/integrate-verify1.md) found:\n${open.filter((p) => p.includes(`[${k}]`)).join('\n')}\nFix each, prove each with a test that fails without the fix, and report as before (write the report to docs/reports/reviews/g4-wave-6a/integrate-repair-${k}.md).`, { label: `repair:${k}`, phase: 'Repair', effort: 'high' })))
  rounds.push({ keys, fixes: fixes.map((f) => (f ? String(f) : null)) })
  if (keys.length) {
    verdict = await ag(`${VERIFY(2)}\n\nREPAIR ROUND: the earlier verifier's BLOCKER/MAJOR problems were:\n${open.join('\n')}\nRepair reports:\n${fixes.map((f, i) => `=== ${keys[i]} ===\n${f ? String(f) : 'NO REPORT'}`).join('\n\n')}\nFor EACH earlier problem, reproduce it against the current tree and say VERIFIED FIXED or NOT FIXED; then grade anything new as before.`, { label: 'reverify:integrate-6a', phase: 'Repair', schema: VERDICT, effort: 'high' })
    open = blockingOf(verdict)
  }
}
log(`integration 6a: ${open.length ? `${open.length} BLOCKER/MAJOR remain` : 'no BLOCKER/MAJOR remains'}`)
return { reports: { codeMinors: repA ? String(repA) : null, docsA11y: repB ? String(repB) : null }, verdict, rounds, open }
