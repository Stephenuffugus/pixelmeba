const VERDICT = {
  type: 'object',
  properties: { ok: { type: 'boolean' }, problems: { type: 'array', items: { type: 'string' } } },
  required: ['ok', 'problems'],
}

const serious = (v) => (v && Array.isArray(v.problems) ? v.problems.filter((p) => /^(BLOCKER|MAJOR|MINOR)/.test(p.trim())) : [])

return await pipeline(
  TASKS,
  (t) =>
    agent(
      `${PRE}\n\nYour Playwright port is ${t.port}.\n\n${t.finish ? FINISH_NOTE(t.finish) + '\n\n' : ''}${t.prompt}\n\n${REPORT(t.key, 'build')} ${BUILD_FINAL}`,
      { label: `${t.finish ? 'finish' : 'build'}:${t.key}`, phase: 'Build' },
    ),
  async (built, t) => {
    const lenses = t.key === 'art-marks' ? [LENSES[0]] : LENSES
    const verdicts = await parallel(
      lenses.map((l) => () => agent(VERIFY(t, built, l), { label: `verify:${t.key}:${l.key}`, phase: 'Verify', schema: VERDICT })),
    )
    return { built, verdicts: verdicts.filter(Boolean) }
  },
  async (v, t) => {
    const rounds = []
    let problems = v.verdicts.flatMap(serious)
    let ok = v.verdicts.length > 0 && v.verdicts.every((x) => x.ok) && problems.filter((p) => !/^MINOR/.test(p.trim())).length === 0
    for (let round = 1; round <= 2 && problems.length > 0; round++) {
      const fixReport = await agent(FIX(t, problems, round), { label: `fix${round}:${t.key}`, phase: 'Fix' })
      const rv = await agent(REVERIFY(t, String(fixReport), problems, round), { label: `reverify${round}:${t.key}`, phase: 'Fix', schema: VERDICT })
      rounds.push({ round, fixReport, reverify: rv })
      if (!rv) break
      const remaining = serious(rv)
      ok = rv.ok
      if (rv.ok && remaining.filter((p) => !/^MINOR/.test(p.trim())).length === 0) {
        problems = remaining
        break
      }
      problems = remaining
    }
    log(`${t.key}: ${ok ? 'clean' : 'problems remain'} after ${rounds.length} fix round(s)`)
    return { key: t.key, built: v.built, verdicts: v.verdicts, rounds, ok, remaining: problems }
  },
