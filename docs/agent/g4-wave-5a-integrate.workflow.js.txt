const VERDICT = {
  type: 'object',
  properties: { ok: { type: 'boolean' }, problems: { type: 'array', items: { type: 'string' } } },
  required: ['ok', 'problems'],
}

phase('Fix')
const [repA, repB] = await parallel([
  () => ag(FIX_A, { label: 'fix:code-minors', phase: 'Fix', effort: 'high' }),
  () => ag(FIX_B, { label: 'fix:e2e-docs', phase: 'Fix', effort: 'high' }),
])
const reports = `=== FIXER code-minors ===\n${repA ? String(repA) : 'NO REPORT'}\n\n=== FIXER e2e-docs ===\n${repB ? String(repB) : 'NO REPORT'}`

const VERIFY = (round) => `You are the adversarial verifier of the lead's integration of Phase 4 wave 5a in /workspaces/pixelmeba (Pixelmeba; follow CLAUDE.md). Do NOT edit any repository file except your report docs/reports/reviews/g4-wave-5a/integrate-verify${round}.md; do NOT run git add/commit/stash/checkout/reset/restore. Scratch in tmp/integrate-verify${round}/. Port 4391 with build folder tmp/dist-integrate-verify (stop the port before and after; never pkill). Two CPUs shared: targeted runs only; never end your turn while a command you started is running.
The integration changes are exactly \`git diff e96976b\` (wip/g4-wave-5a, the backup of the wave 5a tree before integration), ignoring docs/DECISIONS.md and docs/agent/** (the lead's). The two fixers' assignments:
--- code-minors ---
${FIX_A}
--- e2e-docs ---
${FIX_B}
--- the fixers' reports ---
${reports}
Try to REFUTE that each item is done correctly, completely and honestly: read the diff; for each code fix write a load-time mutant under tmp/ (or revert the hunk in a scratch copy) and show the new test fails without it; check D08/D09 for the notification rule; check the duplicate fix against how relationships reach the UI; check that the ARCH text matches src/worker/protocol.ts, notebook.ts and saveFile.ts exactly (quote mismatches); re-run the changed unit tests, tests/fixtures/g2-replay.test.ts and tests/fixtures/trajectory-fence, and the changed e2e tests on desktop only (the fixer ran phone-portrait; re-run one phone-portrait test only if a selector looks layout-dependent). Grade every problem exactly like "MAJOR [code-minors] <file:line> — <what is wrong> — <repro>" (keys: code-minors, e2e-docs, lead); BLOCKER/MAJOR only for real defects, MINOR/NIT otherwise; most severe first. Write the same list to the report file.`

phase('Verify')
let verdict = await ag(VERIFY(1), { label: 'verify:integrate', phase: 'Verify', schema: VERDICT, effort: 'high' })
const blockingOf = (v) => (v && Array.isArray(v.problems) ? v.problems.filter((p) => /^\s*(BLOCKER|MAJOR)/.test(p)) : [])
const rounds = []
let open = blockingOf(verdict)
if (open.length) {
  phase('Repair')
  const keys = ['code-minors', 'e2e-docs'].filter((k) => open.some((p) => p.includes(`[${k}]`)))
  log(`repair: ${keys.join(', ') || 'lead-only problems'}`)
  const fixes = await parallel(keys.map((k) => () => ag(`${k === 'code-minors' ? FIX_A : FIX_B}\n\nYou are the repair round for this assignment (the work is in the tree; do not start over). The adversarial verifier found:\n${open.filter((p) => p.includes(`[${k}]`)).join('\n')}\nFix each, prove each with a test that fails without the fix, and report as before.`, { label: `repair:${k}`, phase: 'Repair', effort: 'high' })))
  rounds.push({ keys, fixes: fixes.map((f) => (f ? String(f) : null)) })
  if (keys.length) {
    verdict = await ag(`${VERIFY(2)}\n\nREPAIR ROUND: the earlier verifier's BLOCKER/MAJOR problems were:\n${open.join('\n')}\nRepair reports:\n${fixes.map((f, i) => `=== ${keys[i]} ===\n${f ? String(f) : 'NO REPORT'}`).join('\n\n')}\nFor EACH earlier problem, reproduce it against the current tree and say VERIFIED FIXED or NOT FIXED; then grade anything new as before.`, { label: 'reverify:integrate', phase: 'Repair', schema: VERDICT, effort: 'high' })
    open = blockingOf(verdict)
  }
}
log(`integration: ${open.length ? `${open.length} BLOCKER/MAJOR remain` : 'no BLOCKER/MAJOR remains'}`)
return { reports: { codeMinors: repA ? String(repA) : null, e2eDocs: repB ? String(repB) : null }, verdict, rounds, open }
