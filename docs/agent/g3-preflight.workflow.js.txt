phase('Build')
const built = await agent(TASK, { label: 'build:preflight', phase: 'Build' })
phase('Verify')
const v = await agent(VERIFY(built), { label: 'verify:preflight:rules', phase: 'Verify', schema: VERDICT })
let problems = serious(v)
let ok = !!v && v.ok && blocking(problems).length === 0
const rounds = []
phase('Fix')
for (let round = 1; round <= 2 && problems.length > 0 && !ok; round++) {
  const fixReport = await agent(FIX(problems, round), { label: `fix${round}:preflight`, phase: 'Fix' })
  const rv = await agent(REVERIFY(String(fixReport), problems, round), { label: `reverify${round}:preflight`, phase: 'Fix', schema: VERDICT })
  rounds.push({ round, fixReport, reverify: rv })
  if (!rv) break
  problems = serious(rv)
  ok = rv.ok && blocking(problems).length === 0
}
log(`preflight: ${ok ? 'clean' : 'problems remain'} after ${rounds.length} fix round(s); ${problems.length} open item(s)`)
return { built, verdict: v, rounds, ok, remaining: problems }
